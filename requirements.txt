from abc import ABC, abstractmethod
from typing import Any, Dict, Optional


class BasePlugin(ABC):
    """Abstract base class for all plugins."""

    @property
    @abstractmethod
    def name(self) -> str: ...

    @abstractmethod
    def execute(self, context: Dict[str, Any]) -> Optional[Any]: ...


class PluginRegistry:
    """Stores and retrieves plugin instances by identifier."""

    _plugins: Dict[str, BasePlugin] = {}

    @classmethod
    def register(cls, plugin: BasePlugin) -> None:
        cls._plugins[plugin.name] = plugin

    @classmethod
    def get(cls, name: str) -> Optional[BasePlugin]:
        return cls._plugins.get(name)


class PluginManager(ABC):
    """Manages lifecycle of registered plugins."""

    registry: PluginRegistry

    @abstractmethod
    def load_all(self) -> Dict[str, BasePlugin]: ...

    @abstractmethod
    def unload(self, name: str) -> bool: ...


__all__: List[str] = ["BasePlugin", "PluginRegistry", "PluginManager"]

from typing import List, Sequence


def normalize_vector(vec: Sequence[float]) -> List[float]:
    """Return unit-normalized copy of input vector."""
    norm = sum(v * v for v in vec) ** 0.5
    if norm == 0:
        return [float(v) for v in vec]
    scale = 1 / norm
    return [v * scale for v in vec]


def cosine_similarity(a: Sequence[float], b: Sequence[float]) -> float:
    """Compute dot product-based similarity between vectors."""
    if len(a) != len(b):
        raise ValueError("Vector length mismatch")
    na = normalize_vector(a)
    nb = normalize_vector(b)
    return sum(x * y for x, y in zip(na, nb))


__all__: List[str] = ["normalize_vector", "cosine_similarity"]

from typing import Optional
from pathlib import Path


def safe_join(base_dir: str, target_path: str) -> Optional[Path]:
    """Join paths safely ensuring containment within base directory."""
    try:
        resolved_base = Path(base_dir).resolve()
        joined = (resolved_base / target_path).resolve()
        if not joined.is_relative_to(resolved_base):
            return None
        return joined
    except OSError:
        return None


def to_windows_path(path_str: str) -> Optional[Path]:
    """Convert string path into Windows-style Path object."""
    try:
        return Path(path_str).resolve()
    except OSError:
        return None


__all__: List[str] = ["safe_join", "to_windows_path"]
